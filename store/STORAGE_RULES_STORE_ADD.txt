match /storeProtected/{productId}/{fileName} {
  allow write: if request.auth != null
    && request.auth.token.email == 'maleksameh121@gmail.com'
    && request.auth.token.email_verified == true
    && request.resource.size < 30 * 1024 * 1024
    && request.resource.contentType == 'application/pdf';
  allow read: if request.auth != null
    && (
      (request.auth.token.email == 'maleksameh121@gmail.com' && request.auth.token.email_verified == true)
      ||
      (firestore.exists(/databases/(default)/documents/storeEntitlements/$(productId + '_' + request.auth.uid))
       && firestore.get(/databases/(default)/documents/storeEntitlements/$(productId + '_' + request.auth.uid)).data.active == true)
    );
}

match /storeProofs/{uid}/{orderId} {
  allow create: if request.auth != null

    && request.resource.size < 8 * 1024 * 1024

  allow read: if request.auth != null
    && request.auth.token.email == 'maleksameh121@gmail.com'
    && request.auth.token.email_verified == true;
  allow update, delete: if false;
}
